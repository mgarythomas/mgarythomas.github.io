{{/* Hugo Render Hook for Markdown Images supporting Page Bundles */}} {{- $src := .Destination -}} {{- $u := urls.Parse $src -}} {{- $img := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $page := .Page -}} {{- with .PageInner }}{{ $page = . }}{{ end -}} {{- $resource := or ($page.Resources.Get $path) ($page.Resources.GetMatch $path) ($page.Resources.GetMatch (printf "*%s*" $path)) ($page.Resources.GetMatch (printf "**/%s" $path)) (resources.Get $path) -}} {{- if $resource -}} {{- $img = $resource.RelPermalink -}} {{- else if strings.HasPrefix $path "/" -}} {{- $img = $path | relURL -}} {{- else -}} {{- $img = printf "%s%s" $page.RelPermalink $path -}} {{- end -}} {{- with $u.RawQuery -}} {{- $img = printf "%s?%s" $img . -}} {{- end -}} {{- with $u.Fragment -}} {{- $img = printf "%s#%s" $img . -}} {{- end -}} {{- else -}} {{- $img = $src -}} {{- end -}}
